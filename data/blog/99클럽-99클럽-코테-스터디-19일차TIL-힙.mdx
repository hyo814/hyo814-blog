```js
const fs = require('fs');
const input = fs.readFileSync('/dev/stdin').toString().trim().split('\n').map(Number);

class AbsHeap {
    constructor() {
        this.heap = [];
    }
    
    push(val) {
        this.heap.push(val);
        this.heap.sort((a, b) => Math.abs(a) - Math.abs(b) || a - b);
    }
    
    pop() {
        return this.heap.length ? this.heap.shift() : 0;
    }
}

const absHeap = new AbsHeap();
const result = [];

for (let i = 1; i < input.length; i++) {
    if (input[i] === 0) {
        result.push(absHeap.pop());
    } else {
        absHeap.push(input[i]);
    }
}

console.log(result.join('\n'));
```
