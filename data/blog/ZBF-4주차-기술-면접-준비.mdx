```js
// 1. Promise 생성
function fetchData() {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      const success = true; // 이 값을 false로 바꿔보세요.
      if (success) {
        resolve("데이터 가져오기 성공!");
      } else {
        reject("데이터 가져오기 실패!");
      }
    }, 2000);
  });
}

// 2. Promise 사용
fetchData()
  .then((message) => {
    console.log(message); // 데이터 가져오기 성공!
  })
  .catch((error) => {
    console.error(error); // 데이터 가져오기 실패!
  });
```

```js
// 1. Promise 생성 (이전 예제와 동일)
function fetchData() {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      const success = true; // 이 값을 false로 바꿔보세요.
      if (success) {
        resolve("데이터 가져오기 성공!");
      } else {
        reject("데이터 가져오기 실패!");
      }
    }, 2000);
  });
}

// 2. Async/Await 사용
async function getData() {
  try {
    const message = await fetchData();
    console.log(message); // 데이터 가져오기 성공!
  } catch (error) {
    console.error(error); // 데이터 가져오기 실패!
  }
}

// 함수 호출
getData();
```
