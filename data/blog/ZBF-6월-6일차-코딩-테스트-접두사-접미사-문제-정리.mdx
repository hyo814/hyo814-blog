2. 접미사 배열(Suffix Array) 생성

```js
// 접두사 배열 생성 함수
function createPrefixArray(str) {
let prefixArray = [];
for (let i = 1; i <= str.length; i++) {
prefixArray.push(str.substring(0, i));
}
return prefixArray;
}
```

```js
//  접미사 배열 생성
function createSuffixArray(str) {
let suffixArray = [];
for (let i = 0; i < str.length; i++) {
suffixArray.push(str.substring(i));
}
return suffixArray;
}
```
