# 99클럽 코테 스터디 27일차 TIL + 오늘의 학습 키워드: 완전 탐색

> 이 글은 [네이버 블로그](https://blog.naver.com/ggamjige8888/223787178342)에 2025년 3월 7일에 올렸던 것을 그대로 옮겨온 것입니다.

- 공부한 내용 본인의 언어로 정리하기

```js
const readline = require("readline");

const rl = readline.createInterface({
    input: process.stdin,
    output: process.output,
});

let count = 0;
rl.on("line", (line) => {
    if (!count) {
        count = Number(line);
    } else {
        main(line);
        rl.close();
    }
}).on("close", () => {
    process.exit();
});

const main = (line) => {
    const numbers = line.split(" ").map(Number).reverse();

    const DP = [];

    for (let i = 0; i < numbers.length; i++) {
        DP[i] = 1;
        for (let j = 0; j < i; j++) {
            if (DP[j] + 1 > DP[i] && numbers[j] < numbers[i]) {
                DP[i] = DP[j] + 1;
            }
        }
    }

    console.log(Math.max(...DP));
};
```
