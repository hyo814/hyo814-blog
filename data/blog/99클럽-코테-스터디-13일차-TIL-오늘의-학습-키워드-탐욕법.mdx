```js
function minActions(N) {
  let actions = 0;
  while (N > 0) {
    if (N % 2 === 1) {
      actions += 1;  // 1비트가 있으면 생성 마법을 사용
    }
    N = Math.floor(N / 2);  // 복제 마법을 위해 오른쪽으로 이동
    if (N > 0) {
      actions += 1;  // 복제 마법의 카운트
    }
  }
  return actions;
}
```
