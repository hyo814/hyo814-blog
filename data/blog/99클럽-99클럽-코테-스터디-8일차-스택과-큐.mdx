```js
function digitCount(num) {
  const count = Array(10).fill(0);
  
  for (const digit of num) {
    count[Number(digit)]++;
  }

  for (let i = 0; i < num.length; i++) {
    if (Number(num[i]) !== count[i]) {
      return false;
    }
  }

  return true;
}
```
